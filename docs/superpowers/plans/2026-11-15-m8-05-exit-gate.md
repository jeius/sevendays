  ```python
  #!/usr/bin/env python3
  """M8 #289 exit-gate extractor: wrangler tail (pretty-printed multi-line JSON)
  -> per-route access rows.

  Reads the capture file (argv[1]). The tail stream is pretty-printed
  multi-line JSON, not JSONL — brace-split parse it (track depth, collect
  top-level {...} objects, quote-aware; the #280 method, per the #285
  execution corrections). The api's pino access lines ride inside
  logs[].message as a LIST of strings — each element one pino JSON string
  containing "evt":"access". Prints one line per route (sorted): hit count,
  statuses, durations, median; then an ALL aggregate. Route = the api access
  line's route field (route pattern; raw path on unmatched routes)."""
  import json
  import statistics
  import sys


  def top_level_objects(text):
      depth = 0
      start = None
      in_str = False
      esc = False
      for i, ch in enumerate(text):
          if in_str:
              if esc:
                  esc = False
              elif ch == "\\":
                  esc = True
              elif ch == '"':
                  in_str = False
              continue
          if ch == '"':
              in_str = True
          elif ch == "{":
              if depth == 0:
                  start = i
              depth += 1
          elif ch == "}":
              depth -= 1
              if depth == 0 and start is not None:
                  yield text[start : i + 1]
                  start = None


  by_route = {}
  statuses = {}
  with open(sys.argv[1], encoding="utf-8", errors="replace") as fh:
      for blob in top_level_objects(fh.read()):
          try:
              rec = json.loads(blob)
          except ValueError:
              continue
          for lg in rec.get("logs") or []:
              msgs = lg.get("message")
              if not isinstance(msgs, list):
                  msgs = [msgs]
              for msg in msgs:
                  if not isinstance(msg, str) or '"evt":"access"' not in msg:
                      continue
                  try:
                      evt = json.loads(msg)
                  except ValueError:
                      continue
                  if evt.get("evt") != "access":
                      continue
                  route = evt.get("route", "?")
                  by_route.setdefault(route, []).append(int(evt.get("durationMs", 0)))
                  statuses.setdefault(route, set()).add(evt.get("status"))

  for route in sorted(by_route):
      ds = sorted(by_route[route])
      st = ",".join(str(s) for s in sorted(statuses[route]))
      print(f"{route}\tn={len(ds)} status={st} durations={ds} median={statistics.median(ds):.0f}")

  all_ds = [d for ds in by_route.values() for d in ds]
  if all_ds:
      print(f"ALL\tn={len(all_ds)} median={statistics.median(all_ds):.0f}")
  ```
